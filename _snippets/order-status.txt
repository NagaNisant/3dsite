export const DESIGN_STATUS_FLOW: OrderStatus[] = [

  "DESIGN_IN_PROGRESS",

];

export const STATUS_LABELS: Record<OrderStatus, string> = {
  QUOTE_REQUESTED: "Quote requested",
  UNDER_REVIEW: "Under review",
  QUOTE_SENT: "Quote sent",
  CUSTOMER_APPROVAL: "Awaiting your approval",
  DESIGN_IN_PROGRESS: "Design in progress",

  QUALITY_CHECK: "Quality check",

  CANCELLED: "Cancelled",
};

export function statusFlowFor(type: OrderType) {
  return type === "design" ? DESIGN_STATUS_FLOW : PRINT_STATUS_FLOW;
}

export function statusIndex(type: OrderType, status: OrderStatus) {
  return statusFlowFor(type).indexOf(status);
}
